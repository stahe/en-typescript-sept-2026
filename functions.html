    8. Functions

    The scripts in this chapter are located in the [fonctions] folder of the project.

    8.1. script [func-01]

    This script examines how parameters are passed to a function: by value for numbers, strings, and booleans; by reference for arrays, literal objects, and functions.

      1  'use strict';
  2  // function parameter passing mode
  3  // -----------------------number - pass-by-value
  4  function doSomethingWithNumber(param: number): void {
  5    param++;
  6    console.log("[param inside function]=", param, "[type]=", typeof (param), "[passage par référence]=", param === count);
  7  }
  8  // call code
  9  let count: number = 10;
 10  doSomethingWithNumber(count);
 11  console.log("[count outside function]=", count);
 12  
 13  // --------------------- string - passed by value
 14  function doSomethingWithString(param: string): void {
 15    param += " xyz"
 16    console.log("[param inside function]=", param, "[type]=", typeof (param), "[passage par référence]=", param === text);
 17  }
 18  // call code
 19  let text: string = "abcd";
 20  doSomethingWithString(text);
 21  console.log("[text outside function]=", text);
 22  
 23  // --------------------- Boolean - passed by value
 24  function doSomethingWithBoolean(param: boolean): void {
 25    param = !param;
 26    console.log("[param inside function]=", param, "[type]=", typeof (param), "[passage par référence]=", param === bool);
 27  }
 28  // call code
 29  let bool: boolean = true;
 30  doSomethingWithBoolean(bool);
 31  console.log("bool [outside function]=", bool);
 32  
 33  // --------------------- array - passed by reference
 34  function doSomethingWithArray(param: number[]): void {
 35    param.push(1000);
 36    console.log("[param inside function]=", param, "[type]=", typeof (param), "[passage par référence]=", param === tab);
 37  }
 38  // call code
 39  const tab: number[] = [10, 20, 30];
 40  doSomethingWithArray(tab);
 41  console.log("[tab outside function]=", tab);
 42  
 43  // --------------------- object - passed by reference
 44  // [any] to be able to dynamically add [unePropriétéNouvelle], as in JS
 45  function doSomethingWithObject(param: any): void {
 46    param.unePropriétéNouvelle = "xyz";
 47    console.log("[param inside function]=", param, "[type]=", typeof (param), "[passage par référence]=", param === obj);
 48  }
 49  // call code
 50  const obj: any = [10, 20, 30];
 51  doSomethingWithObject(obj);
 52  console.log("[obj outside function]=", obj);
 53  
 54  // --------------------- function - passed by reference
 55  // [any] because a property is dynamically assigned to the function itself (see below)
 56  function doSomethingWithFunction(param: any): void {
 57    // something rather odd, but it works anyway
 58    param.unePropriétéNouvelle = "xyz";
 59    console.log("[param inside function]=", param, "[type]=", typeof (param), "[passage par référence]=", param === f);
 60  }
 61  // call code
 62  const f: any = (x: number) => x + 4;
 63  doSomethingWithFunction(f);
 64  console.log("[f outside function]=", f, f.unePropriétéNouvelle, typeof (f));
    • Primitive types (number, string, boolean) are passed by value: modifying param inside the function never changes the original variable (pass-by-reference = false in each case);
    • arrays, objects, and functions are passed by reference: param === tab (or obj, or f) evaluates to true, and a modification made inside the function is visible outside;
    • line 41: function doSomethingWithObject(param: any) — typed as any because a property (unePropriétéNouvelle) not provided for in the original type of the obj array is dynamically added;
    • line 51: same thing for doSomethingWithFunction — a property is added to the function itself; this behavior (JavaScript) is a bit confusing but perfectly valid (a function is also an object).

  3  [param inside function]= abcd xyz [type]= string [passage par référence]= false
  4  [text outside function]= abcd
  5  [param inside function]= false [type]= boolean [passage par référence]= false
  6  bool [outside function]= true
  7  [param inside function]= [ 10, 20, 30, 1000 ] [type]= object [passage par référence]= true
  8  [tab outside function]= [ 10, 20, 30, 1000 ]
  9  [param inside function]= [ 10, 20, 30, 'unePropriétéNouvelle': 'xyz' ] [type]= object [passage par référence]= true

 11  [param inside function]= [Function: f] { 'unePropriétéNouvelle': 'xyz' } [type]= function [passage par référence]= true
 12  [f outside function]= [Function: f] { 'unePropriétéNouvelle': 'xyz' } xyz function

    8.2. script [func-02]

    This script shows that [function] is a data type like any other: a variable can have this type, and there are two ways to define a function—using the function keyword or the arrow notation =>.

      1  'use strict';
  2  // You can assign a function to a variable
  3  const variable1: (a: number, b: number) => number = function (a, b) {
  4    return a + b;
  5  };
  6  console.log("typeof(variable1)=", typeof (variable1));
  7  // the variable can then be used as a function
  8  console.log("variable1(10,12)=", variable1(10, 12));
  9  // the function can be defined using the => notation
 10  const variable2: (a: number, b: number, c: number) => number = (a, b, c) => {
 11    return a - b + c;
 12  };
 13  console.log("variable2(10,12,14)=", variable2(10, 12, 14));
 14  // You can omit the curly braces if there is only one expression in the function body
 15  // This expression is then the function’s return value
 16  const variable3 = (a: number, b: number, c: number): number => a + b + c;
 17  console.log("variable3(10,12,14)=", variable3(10, 12, 14));
    • line 3: const variable1: (a: number, b: number) => number = function (a, b) {...} — the type of a function variable is declared as a signature (parameters) => typeDeRetour;
    • Lines 10–12: The same thing using arrow notation: (a, b, c) => { ... };
    • line 16: without curly braces, the function body is reduced to a single expression, which automatically becomes the return value — (a, b, c): number => a + b + c.
    npx tsx fonctions/func-02.ts

    Execution result:

      1  typeof(variable1)= function
  2  variable1(10,12)= 22
  3  variable2(10,12,14)= 12
  4  variable3(10,12,14)= 36

    8.3. script [func-03]

    This script demonstrates that a function can be passed as an argument to another function—a technique widely used in the TypeScript/JavaScript frameworks (callbacks, map/filter/reduce from the “Arrays” chapter...):

      1  'use strict';
  2  // A function's parameters can be of type [fonction]
  3  
  4  // function f1
  5  function f1(param1: number, param2: number): number {
  6    return param1 + param2 + 10;
  7  }
  8  // function f2
  9  function f2(param1: number, param2: number): number {
 10    return param1 + param2 + 20;
 11  }
 12  // function g with function f as a parameter
 13  function g(param1: number, param2: number, f: (a: number, b: number) => number): number {
 14    return f(param1, param2) + 100;
 15  }
 16  // Uses of g
 17  console.log(g(0, 10, f1));
 18  console.log(g(0, 10, f2));
 19  // the actual function-type parameter can be passed by reference—form 1
 20  console.log(g(0, 10, (param1, param2) => {
 21    return param1 + param2 + 30;
 22  }));
 23  // The actual function-type parameter can be passed by reference—form 2
 24  console.log(g(0, 10, function (param1, param2) {
 25    return param1 + param2 + 40;
 26  }));
    • line 13: function g(param1: number, param2: number, f: (a: number, b: number) => number): number — the third parameter f is itself typed as a function;
    • lines 20–22 and 24–26: the actual function-type parameter can be passed directly, either as an arrow function or a regular function, without first being stored in a variable.
    npx tsx fonctions/func-03.ts

    Execution result:

      1  120
  2  130
  3  140
  4  150

    8.4. script [func-04]

    This script demonstrates that a function can behave like a class—a mechanism that predates ES6 but is still valid, although it has now been replaced by the class keyword (see the chapter “Classes”):

      1  'use strict';
  2  // A function can be used as an object
  3  
  4  // An empty shell
  5  // [any]: The type of [this] is intentionally flexible here, because we assign
  6  // properties to the function itself, APRÈS, in its definition (see below) —
  7  // a schema that TypeScript cannot deduce on its own
  8  function f(this: any): void {
  9  
 10  }
 11  // to which properties are assigned from the outside
 12  (f as any).prop1 = "val1";
 13  (f as any).show = function (this: any) {
 14    console.log(this.prop1);
 15  };
 16  // use of f
 17  (f as any).show();
 18  
 19  // a function g that behaves like a class
 20  function g(this: any): void {
 21    this.prop2 = "val2";
 22    this.show = function (this: any) {
 23      console.log(this.prop2);
 24    };
 25  }
 26  // instantiation of the function with [new]
 27  new (g as any)().show();
    • lines 5–9: f is an “empty shell” function, typed as this: any—this special annotation for the first parameter tells TypeScript the type of this inside the function, which is necessary here because properties are added to it from the outside, a pattern that TypeScript cannot deduce on its own;
    • Lines 12–15: Properties are assigned to f from outside—(f as any).prop1 = ...;
    • line 17: f.show() is called, not f()—this is indeed the notation for using an object, not a traditional function call;
    • lines 20–27: g defines this.prop2 and this.show within itself, just as a class constructor would;
    • line 29: new (g as any)() **instantiates **g as a class.
    npx tsx fonctions/func-04.ts

    Execution result:

      1  val1
  2  val2
    Note: ES6 introduced the `class` keyword (see the chapter “Classes”), which allows you to achieve this result directly without using this functional workaround.

    8.5. script [func-05]

    This script demonstrates the use of [rest operator] (...), which collects all remaining arguments from a call into an array:

      1  'use strict';
  2  // rest operator
  3  function f(arg1: number, ...otherArgs: unknown[]): void {
  4    // first argument
  5    console.log("arg1=", arg1);
  6    // the other arguments
  7    let i = 0;
  8    otherArgs.forEach(element => {
  9      console.log("otherArguments[", i, "]=", element);
 10      i++;
 11    });
 12  }
 13  
 14  // call
 15  f(1, "deux", "trois", { x: 2, y: 3 })
    • Line 3: function f(arg1: number, ...otherArgs: unknown[]): void — otherArgs groups all arguments passed after arg1, regardless of their number. The unknown[] type is appropriate here because otherArgs intentionally mixes different types (strings and objects, in the call on line 15).
    npx tsx fonctions/func-05.ts

    Execution result:

      1  arg1= 1
  2  otherArguments[ 0 ]= deux
  3  otherArguments[ 1 ]= trois
  4  otherArguments[ 2 ]= { x: 2, y: 3 }
